function analyzeImage(imagesBase64, clientEntities, ctx) {
  var parts = [{ text: buildPrompt(clientEntities, ctx) }];
  imagesBase64.forEach(function (b64) {
    parts.push({ inline_data: { mime_type: 'image/jpeg', data: b64 } });
  });
  // V23: thử model mới trước, lỗi/hết quota thì tự rơi về model dự phòng
  var payload = JSON.stringify({
    contents: [{ parts: parts }],
    generationConfig: { temperature: 0.1, response_mime_type: 'application/json' }
  });
  var models = [GEMINI_MODEL, GEMINI_MODEL_FALLBACK];
  var response = null, lastErr = '';
  for (var mi = 0; mi < models.length; mi++) {
    response = UrlFetchApp.fetch(
      'https://generativelanguage.googleapis.com/v1beta/models/' + models[mi] + ':generateContent?key=' + GEMINI_API_KEY,
      { method: 'post', contentType: 'application/json', muteHttpExceptions: true, payload: payload }
    );
    if (response.getResponseCode() === 200) break;
    lastErr = models[mi] + ' → HTTP ' + response.getResponseCode() + ': ' + response.getContentText().slice(0, 160);
  }
  if (response.getResponseCode() !== 200) throw new Error('Gemini lỗi: ' + lastErr);

  // V44: parse phòng thủ — Gemini trả 200 nhưng thiếu candidates (bị chặn an toàn) hoặc JSON dính rác
  // sẽ KHÔNG còn ném TypeError/SyntaxError làm chết hàng đợi; trả emptyAnalysis (báo cáo vẫn lưu).
  var _r0; try { _r0 = JSON.parse(response.getContentText()); } catch (_e0) { return emptyAnalysis(); }
  var _c0 = _r0 && _r0.candidates && _r0.candidates[0];
  var _p0 = _c0 && _c0.content && _c0.content.parts;
  var text = (_p0 && _p0[0] && _p0[0].text) ? _p0[0].text : '';
  if (!text) return emptyAnalysis();
  var json = safeParseJson(text);
  if (!json) return emptyAnalysis();

  // Parse phòng thủ: mọi field đều có fallback
  var ocr = json.ocr || {};
  return {
    category: json.category || '',
    description: json.description || '',
    rawText: ocr.raw_text || '',
    entities: (ocr.entities || []).map(function (en) {
      return { type: en.type || 'freeText', rawValue: en.raw_value || '',
               normalizedValue: en.normalized_value || en.raw_value || '',
               confidence: Number(en.confidence) || 0, source: 'cloud' };
    }),
    issues: json.issues || [],
    conflicts: json.conflicts || [],
    uncertain: json.uncertain_notes || '',
    confidence: Number(json.overall_confidence) || 0
  };
}

// V44: tách JSON an toàn khỏi câu trả lời Gemini (bỏ ```fences```, cắt từ { đầu tới } cân bằng cuối,
// bỏ qua dấu ngoặc nằm trong chuỗi). Hỏng → trả null để hàm gọi rơi về emptyAnalysis (không ném lỗi).
function safeParseJson(text){
  if(!text) return null;
  var t=String(text).replace(/```json|```/g,'').trim();
  try{ return JSON.parse(t); }catch(e){}
  var a=t.indexOf('{'); if(a<0) return null;
  var depth=0,inStr=false,esc=false;
  for(var i=a;i<t.length;i++){
    var c=t.charAt(i);
    if(esc){esc=false;continue;}
    if(c==='\\'){esc=true;continue;}
    if(c==='"') inStr=!inStr;
    if(inStr) continue;
    if(c==='{') depth++;
    else if(c==='}'){ depth--; if(depth===0){ try{ return JSON.parse(t.slice(a,i+1)); }catch(e2){ return null; } } }
  }
  return null;
}

  g.forEach(function (r) { var m = String(r[0]).match(/(\d+)/); if (m) { var v = parseInt(m[1], 10); if (v > mx) mx = v; } });
  return mx + 1;
}

/** "SECTION 19" / "Sec 19" / "S19" / 19 → "19". Không có số thì giữ nguyên chữ. */
function normSection(v) {
  var t = String(v == null ? '' : v).trim();
  if (!t) return '';
  var m = t.replace(/^\s*(section|sec|s)\s*[.:#-]?\s*/i, '').trim();

/** Đọc danh mục → {projects, tree{project:{section:[part]}}, descs}. Có cache 5 phút cho app tải nhanh. */
function readCatalog(force) {
  var ck = 'catalog_v1';
  if (!force) {
    try { var hit = CacheService.getScriptCache().get(ck); if (hit) return JSON.parse(hit); } catch (eC) {}
  }
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var t = ensureCatalog(ss);
  var out = { projects: [], tree: {}, descs: [], descg: [] };
  if (t.cat.getLastRow() >= 2) {
    var cv = t.cat.getRange(2, 1, t.cat.getLastRow() - 1, 3).getValues();
    var lastPj = '';
    for (var i = 0; i < cv.length; i++) {
      var pj = String(cv[i][0] || '').trim();
      if (pj) lastPj = pj; else pj = lastPj;          // bỏ trống Project = theo dòng trên (dán cho nhanh)
      var sc = normSection(cv[i][1]);                  // "SECTION 19" / "Sec 19" → "19"
      var pt = String(cv[i][2] || '').trim();
      if (!pj || !pt) continue;
      if (!out.tree[pj]) { out.tree[pj] = {}; out.projects.push(pj); }
      if (!out.tree[pj][sc]) out.tree[pj][sc] = [];
      if (out.tree[pj][sc].indexOf(pt) < 0) out.tree[pj][sc].push(pt);
    }
  }
  if (t.desc.getLastRow() >= 2) {
    var dvv = t.desc.getRange(2, 1, t.desc.getLastRow() - 1, 2).getValues();
    var gIdx = {};
    for (var j = 0; j < dvv.length; j++) {
      var dd = String(dvv[j][0] || '').trim();
      if (!dd || out.descs.indexOf(dd) >= 0) continue;
      out.descs.push(dd);
      var gg = String(dvv[j][1] || '').trim() || 'Khác';
      if (gIdx[gg] === undefined) { gIdx[gg] = out.descg.length; out.descg.push({ g: gg, items: [] }); }
      out.descg[gIdx[gg]].items.push(dd);
    }
  }
  try { CacheService.getScriptCache().put(ck, JSON.stringify(out), 300); } catch (eC2) {}
  return out;
}

// ============ ENTITY MERGE: user > onDevice > cloud ============

function mergeEntities(clientEntities, cloudEntities) {
  var priority = { user: 3, userManual: 3, onDevice: 2, onDeviceMlKit: 2, cloud: 1, cloudGemini: 1 };
  var all = (clientEntities || []).concat(cloudEntities || []);
  var byKey = {};
  all.forEach(function (en) {
    if (!en || !en.type) return;
    var val = String(en.normalizedValue || en.normalized_value || en.rawValue || en.raw_value || '');
    if (!val) return;
    var key = en.type + '|' + val.toLowerCase();
    var p = priority[en.source] || 1;
    if (!byKey[key] || p > byKey[key].p) {
      byKey[key] = { p: p, type: en.type, value: val };
    }
  });
  return Object.keys(byKey).map(function (k) { return byKey[k]; });
}

               .map(function (e) { return e.value; })
               .join(', ');
}

function matchProjectByGps(lat, lng) {
  if (!lat || !lng) return null;
  var best = null, bestDist = Infinity;
  getProjects().forEach(function (p) {
    if (!p.lat || !p.lng) return;
    var d = haversine(lat, lng, p.lat, p.lng);
    if (d <= p.radius && d < bestDist) { best = p.name; bestDist = d; }
  });
  return best;
}

function haversine(lat1, lng1, lat2, lng2) {
  var R = 6371000, rad = Math.PI / 180;
  var dLat = (lat2 - lat1) * rad, dLng = (lng2 - lng1) * rad;
  var a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
          Math.cos(lat1 * rad) * Math.cos(lat2 * rad) * Math.sin(dLng / 2) * Math.sin(dLng / 2);
  return 2 * R * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
}

function getUsers() {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sh = ss.getSheetByName(USERS_TAB);
  if (!sh || sh.getLastRow() < 2) return [];
  var vals = sh.getRange(2, 1, sh.getLastRow() - 1, 7).getValues();
  var out = [];
  vals.forEach(function (r, i) {
    var name = String(r[0] || '').trim().replace(/\s+/g, ' ');
    if (!name) return;
    if (String(r[4] || '').trim()) return;  // có chữ ở cột Ẩn → bỏ qua
    out.push({
      name: name,
      role: String(r[1] || '').trim(),
      pin: String(r[2] || '').trim(),
      aliases: String(r[3] || '').split(',').map(function (s) { return s.trim(); }).filter(Boolean),
      devices: String(r[6] || '').split(',').map(function (s) { return s.trim(); }).filter(Boolean),  // V19
      row: i + 2  // V19
    });
  });
  return out;
}

// V19: tra người theo ID máy
function userByDevice(users, dev) {
  if (!dev) return null;
  for (var i = 0; i < users.length; i++)
    if (users[i].devices && users[i].devices.indexOf(dev) !== -1) return users[i];
  return null;
}

// V19: tên người gửi cho báo cáo — ưu tiên ID máy đã đăng ký, sau đó mới tới tên gõ vào
function senderName(data) {
  var u = userByDevice(getUsers(), String(data.deviceId || '').trim());
  return u ? u.name : canonicalUser(data.user);
}

function listUserNames() {
  return getUsers().map(function (u) { return u.name; });
}

/** Chuẩn hóa để SO SÁNH tên: thường hóa, bỏ dấu tiếng Việt, gộp khoảng trắng. */
var VN_FROM = 'àáạảãâầấậẩẫăằắặẳẵèéẹẻẽêềếệểễìíịỉĩòóọỏõôồốộổỗơờớợởỡùúụủũưừứựửữỳýỵỷỹđ';
var VN_TO   = 'aaaaaaaaaaaaaaaaaeeeeeeeeeeeiiiiiooooooooooooooooouuuuuuuuuuuyyyyyd';
function normName(s) {
  s = String(s || '').trim().toLowerCase().replace(/\s+/g, ' ');
  try { s = s.normalize('NFC'); } catch (e) {}  // gộp dấu rời (nếu bàn phím gõ kiểu tách dấu)
  var out = '';
  for (var i = 0; i < s.length; i++) {
    var idx = VN_FROM.indexOf(s.charAt(i));
    out += idx >= 0 ? VN_TO.charAt(idx) : s.charAt(i);
  }
  return out;
}

/** Tên thô → tên chuẩn theo tab Người dùng (khớp cả biến thể). Không khớp → trả tên đã trim. */
function canonicalUser(raw) {
  var name = String(raw || '').trim().replace(/\s+/g, ' ');
  if (!name) return '';
  var n = normName(name);
  var users = getUsers();
  for (var i = 0; i < users.length; i++) {
    if (normName(users[i].name) === n) return users[i].name;
    for (var j = 0; j < users[i].aliases.length; j++) {
      if (normName(users[i].aliases[j]) === n) return users[i].name;
    }
  }
  return name;
}

function pinConfigured(users) {
  return (users || getUsers()).filter(function (u) { return u.pin; }).length > 0;
}

function validPin(users, pin) {
  pin = String(pin || '').trim();
  if (!pin) return false;
  return (users || getUsers()).filter(function (u) { return u.pin && u.pin === pin; }).length > 0;
}

/** Tạo PDF báo cáo QC: project ('__all__' = tất cả), days (0 = toàn bộ). Trả {url, name}. */
function makePdfReport(project, days) {
  var ss = SpreadsheetApp.openById(SHEET_ID);
  var sh = (project && project !== '__all__') ? ss.getSheetByName(project) : ss.getSheetByName(ALL_TAB);
  if (!sh || sh.getLastRow() < 2) throw new Error('Không có dữ liệu cho dự án này');
  var vals = sh.getRange(2, 1, sh.getLastRow() - 1, 21).getValues();
  var since = (days > 0) ? new Date(Date.now() - days * 86400000) : null;
  var rows = [];
  for (var i = vals.length - 1; i >= 0 && rows.length < 20; i--) {
    var tv = vals[i][0];
    var t = (tv instanceof Date) ? tv : new Date(String(tv).replace(' ', 'T'));
    if (since && !isNaN(t) && t < since) continue;
    rows.push(vals[i]);
  }
  if (!rows.length) throw new Error('Không có báo cáo nào trong kỳ đã chọn');

  function esc(x) { return String(x || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }
  var nWarn = 0, users = {};
  rows.forEach(function (r) { if (String(r[4] || '').trim()) nWarn++; if (r[16]) users[r[16]] = 1; });

  var imgBudget = 24; // tổng số ảnh nhúng tối đa (giữ PDF nhẹ)
  function imgTags(linkCell) {
    var out = '';
    var links = String(linkCell || '').split('\n').filter(Boolean);
    for (var li = 0; li < links.length && li < 2 && imgBudget > 0; li++) {
      try {
        var m = String(links[li]).match(/[-\w]{25,}/);
        if (!m) continue;
        var blob = DriveApp.getFileById(m[0]).getBlob();
        out += '<img src="data:image/jpeg;base64,' + Utilities.base64Encode(blob.getBytes()) +
               '" style="width:175px;margin:3px;border-radius:4px;vertical-align:top">';
        imgBudget--;
      } catch (e) {}
    }
    if (links.length > 2) out += '<div style="font-size:9px;color:#888">(+' + (links.length - 2) + ' ảnh — xem Drive)</div>';
    return out;
  }

  var projLabel = (project && project !== '__all__') ? project : 'Tất cả dự án';
  var kyLabel = days > 0 ? days + ' ngày gần nhất' : 'Toàn bộ dữ liệu';
  var today = Utilities.formatDate(new Date(), 'GMT+7', 'dd/MM/yyyy HH:mm');
  var html = '<html><head><meta charset="UTF-8"><style>' +
    'body{font-family:Arial,sans-serif;font-size:11px;color:#222;margin:24px}' +
    'h1{color:#1a73e8;font-size:20px;margin:0}' +
    '.sub{color:#666;font-size:11px;margin:4px 0 14px}' +
    '.kpi{display:inline-block;background:#e8f0fe;border-radius:8px;padding:8px 16px;margin-right:8px;text-align:center}' +
    '.kpi b{font-size:18px;color:#1a73e8;display:block}' +
    '.rep{border:1px solid #ddd;border-radius:8px;padding:10px;margin-top:12px;page-break-inside:avoid}' +
    '.meta{color:#555;font-size:10px;margin-bottom:4px}' +
    '.warn{background:#fce8e6;color:#c5221f;border-radius:6px;padding:6px;margin-top:4px;white-space:pre-line}' +
    '.desc{color:#333;margin-top:4px}' +
    '.chip{display:inline-block;background:#e8f0fe;color:#1a73e8;border-radius:5px;padding:1px 6px;font-size:10px;font-weight:bold;margin-right:4px}' +
    '</style></head><body>' +
    '<h1>📷 BÁO CÁO QC HIỆN TRƯỜNG</h1>' +
    '<div class="sub">Dự án: <b>' + esc(projLabel) + '</b> &nbsp;•&nbsp; Kỳ: ' + esc(kyLabel) +
    ' &nbsp;•&nbsp; Xuất ngày: ' + today + ' &nbsp;•&nbsp; Hệ thống Punch List</div>' +
    '<div><span class="kpi"><b>' + rows.length + '</b>báo cáo</span>' +
    '<span class="kpi"><b style="color:#c5221f">' + nWarn + '</b>cảnh báo</span>' +
    '<span class="kpi"><b>' + Object.keys(users).length + '</b>người chụp</span></div>';

  rows.forEach(function (r) {
    var unitInfo = [r[18], r[19], r[20]].map(function (x) { return String(x || '').trim(); }).filter(Boolean).join(' / ');
    html += '<div class="rep">' +
      '<div><b>' + esc(unitInfo || r[2] || '(chưa phân loại)') + '</b>' +
      (r[2] ? ' <span class="chip">' + esc(r[2]) + '</span>' : '') + '</div>' +
      '<div class="meta">🕒 ' + esc(fmtCell(r[0])) + (r[16] ? ' • 👤 ' + esc(r[16]) : '') + ' • 📁 ' + esc(r[1]) + '</div>' +
      (r[3] ? '<div class="desc">🤖 ' + esc(r[3]) + '</div>' : '') +
      (String(r[4] || '').trim() ? '<div class="warn">⚠️ ' + esc(r[4]) + '</div>' : '') +
      (r[5] ? '<div class="desc">📝 ' + esc(r[5]) + '</div>' : '') +
      (r[8] ? '<div class="desc">📐 Kích thước: ' + esc(r[8]) + '</div>' : '') +
      '<div>' + imgTags(r[15]) + '</div>' +
      '</div>';
  });
  // V32: khối KÝ NGHIỆM THU (chữ ký Sếp nhúng sẵn nếu có SIGNATURE_FILE_ID)
  var signImg = '';
  try {
    if (SIGNATURE_FILE_ID) {
      var sb = DriveApp.getFileById(SIGNATURE_FILE_ID).getBlob();
      signImg = '<img src="data:image/png;base64,' + Utilities.base64Encode(sb.getBytes()) + '" style="height:60px">';
    }
  } catch (e) {}
  html += '<table style="width:100%;margin-top:28px;border-collapse:collapse;page-break-inside:avoid"><tr>' +
    '<td style="width:50%;text-align:center;vertical-align:bottom;padding:8px">' +
      '<div style="font-weight:bold;font-size:11px">ĐẠI DIỆN QC</div>' +
      '<div style="height:70px;line-height:70px">' + signImg + '</div>' +
      '<div style="border-top:1px solid #999;margin:4px 30px 0;padding-top:4px;color:#555;font-size:10px">Ký, ghi rõ họ tên</div></td>' +
    '<td style="width:50%;text-align:center;vertical-align:bottom;padding:8px">' +
      '<div style="font-weight:bold;font-size:11px">ĐẠI DIỆN KHÁCH HÀNG / TƯ VẤN GIÁM SÁT</div>' +
      '<div style="height:70px"></div>' +
      '<div style="border-top:1px solid #999;margin:4px 30px 0;padding-top:4px;color:#555;font-size:10px">Ký, ghi rõ họ tên</div></td>' +
    '</tr></table>';
  html += '<div style="color:#aaa;font-size:9px;margin-top:16px">Tạo tự động bởi Punch List — ' + today + '</div></body></html>';

  var fname = 'PunchList_' + projLabel.replace(/[^\wÀ-ỹ ]/g, '').replace(/\s+/g, '-') + '_' +
    Utilities.formatDate(new Date(), 'GMT+7', 'yyyyMMdd-HHmm');
  var pdf = Utilities.newBlob(html, 'text/html', fname + '.html').getAs('application/pdf').setName(fname + '.pdf');
  var folder = getOrCreateSubfolder(getOrCreateFolder(FOLDER_NAME), 'Báo cáo PDF');
  var file = folder.createFile(pdf);
  return { url: file.getUrl(), name: file.getName() };
}

/** Gửi tin Telegram khi AI phát hiện cảnh báo. row = 21 cột từ tab Tổng hợp. */
function sendIssueAlert(row, issuesText) {
  if (!TELEGRAM_BOT_TOKEN || !TELEGRAM_CHAT_ID) return;
  function esc(s) { return String(s || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); }
  var time = fmtCell(row[0]), project = String(row[1] || ''), cat = String(row[2] || '');
  var desc = String(row[3] || ''), user = String(row[16] || '');
  var unitInfo = [row[18], row[19], row[20]].map(function (x) { return String(x || '').trim(); })
    .filter(Boolean).join(' / ');
  var imgs = String(row[15] || '').split('\n').filter(Boolean);
  var text =
    '⚠️ <b>CẢNH BÁO — ' + esc(project) + '</b>' + (unitInfo ? '\n🔩 ' + esc(unitInfo) : '') +
    '\n🕒 ' + esc(time) + (user ? ' • 👤 ' + esc(user) : '') +
    (cat ? '\n📂 ' + esc(cat) : '') +
    '\n\n' + esc(issuesText) +
    (desc ? '\n\n🤖 ' + esc(desc) : '') +
    (imgs.length ? '\n\n' + imgs.map(function (u, i) { return '<a href="' + u + '">📷 Ảnh ' + (i + 1) + '</a>'; }).join('  ') : '') +
    '\n<a href="https://docs.google.com/spreadsheets/d/' + SHEET_ID + '">📊 Mở Sheet</a>';
  UrlFetchApp.fetch('https://api.telegram.org/bot' + TELEGRAM_BOT_TOKEN + '/sendMessage', {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify({ chat_id: TELEGRAM_CHAT_ID, text: text, parse_mode: 'HTML', disable_web_page_preview: true }),
    muteHttpExceptions: true
  });
  // V22.1: gửi kèm tối đa 3 ẢNH THẬT vào chat (bot tự đọc từ Drive — không cần mở quyền chia sẻ)
  for (var pi = 0; pi < imgs.length && pi < 3; pi++) {
    try {
      var idm = String(imgs[pi]).match(/[-\w]{25,}/);
      if (!idm) continue;
      var blob = DriveApp.getFileById(idm[0]).getBlob();
      UrlFetchApp.fetch('https://api.telegram.org/bot' + TELEGRAM_BOT_TOKEN + '/sendPhoto', {
        method: 'post',
        payload: {
          chat_id: TELEGRAM_CHAT_ID,
          photo: blob,
          caption: '📷 Ảnh ' + (pi + 1) + '/' + imgs.length +
            (imgs.length > 3 && pi === 2 ? ' (còn ' + (imgs.length - 3) + ' ảnh — xem Sheet)' : '')
        },
        muteHttpExceptions: true
      });
    } catch (e) { Logger.log('sendPhoto lỗi: ' + e); }
  }
}
